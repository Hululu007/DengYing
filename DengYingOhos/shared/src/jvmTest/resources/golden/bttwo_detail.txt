            function updateCount(el, delta) {
                var n = parseInt(el.textContent, 10) || 0;
                el.textContent = Math.max(0, n + delta);
            }

            
            async function initVoteStatus() {
                try {
                    const response = await fetch(`/api/votes/check?vod_id=${vodid}`);
                    const data = await response.json();
                    
                    if (data.code === 200 && data.data) {
                        const { voted, vote_type } = data.data;
                        const likeUse    = likeButton.querySelector('use');
                        const dislikeUse = dislikeButton.querySelector('use');
                        
                        if (voted) {
                            if (vote_type === 1) {
                                likeButton.classList.add('voted');
                                if (likeUse) likeUse.setAttribute('href', '/static/icons/sprite.svg#ri-thumb-up-fill');
                            } else if (vote_type === -1) {
                                dislikeButton.classList.add('voted');
                                if (dislikeUse) dislikeUse.setAttribute('href', '/static/icons/sprite.svg#ri-thumb-down-fill');
                            }
                        }
                    }
                } catch (error) {
                    console.error('获取状态失败:', error);
                }
            }

            
            function onVoteSuccess(voteType, isCancel, wasOpposite) {
                if (isCancel) {
                    
                    voteType === 1 ? updateCount(likeCountEl, -1) : updateCount(dislikeCountEl, -1);
                } else {
                    if (wasOpposite) {
                        
                        voteType === 1
                            ? (updateCount(dislikeCountEl, -1), updateCount(likeCountEl, +1))
                            : (updateCount(likeCountEl, -1),    updateCount(dislikeCountEl, +1));
                    } else {
                        
                        voteType === 1 ? updateCount(likeCountEl, +1) : updateCount(dislikeCountEl, +1);
                    }
                }
            }

            
            if (window.UserAuth && window.UserAuth.isLoggedIn()) {
                initVoteStatus();
            }

                    if (ok === true) onVoteSuccess(1, isCancel, wasOpposite && !isCancel);
                });
            });

                var wasOpposite= likeButton.classList.contains('voted');
                window.handleVote(this, -1, vodid).then(function(ok) {

                });
            });

<script>
    const vodId = 'ch4q7ivpp';


function ratingList() {
    return {
        list: [],
        total: 0,
        page: 1,
        pageSize: 20,
        loading: true,
        loadingMore: false,

        init() {
            this.fetchList(1);
        },

        async fetchList(page) {
            const isFirst = page === 1;
            if (isFirst) {
                this.loading = true;
            } else {
                this.loadingMore = true;
            }

            try {
                const response = await fetch(`/api/ratings/list?vod_id=${vodId}&page=${page}&page_size=${this.pageSize}`);
                const data = await response.json();

                if (data.code === 200 && data.data) {
                    const items = data.data.list || [];
                    this.total = data.data.total || 0;
                    this.page = data.data.page || page;
                    this.list = isFirst ? items : this.list.concat(items);
                }
            } catch (error) {
                console.error('获取评分列表失败:', error);
            } finally {
                this.loading = false;
                this.loadingMore = false;
            }
        },

        loadMore() {
            this.fetchList(this.page + 1);
        },

        
        avatarColor(name) {
            const str = name || '匿';
            let hash = 0;
            for (let i = 0; i < str.length; i++) {
                hash = (hash * 31 + str.charCodeAt(i)) >>> 0;
            }
            const hue = hash % 360;
            
            return `linear-gradient(135deg, hsl(${hue}, 60%, 50%), hsl(${(hue + 28) % 360}, 65%, 42%))`;
        }
    };
}


function ratingModal() {
    return {
        rating: 0,
        hoverRating: 0,
        comment: '',
        errorMessage: '',
        successMessage: '',
        submitting: false,
        userRating: null,
        
        init() {
            
            this.fetchUserRating();
        },
        
        
        async fetchUserRating() {
            try {
                const response = await fetch(`/api/ratings/my-rating?vod_id=${vodId}`);
                const data = await response.json();
                
                if (data.code === 200 && data.data) {
                    this.userRating = data.data;
                    this.rating = data.data.rating;
                    this.comment = data.data.comment || '';
                }
            } catch (error) {
                console.error('获取评分失败:', error);
            }
        },
        
        
        getRatingText(score) {
            if (score === 0) return '请选择评分';
            if (score <= 2) return '很差';
            if (score <= 4) return '较差';
            if (score <= 6) return '一般';
            if (score <= 8) return '不错';
            if (score <= 9) return '很好';
            return '完美';
        },
        
        
        async submitRating() {
            if (this.rating === 0) {
                this.errorMessage = '请选择评分';
                return;
            }
            
            this.errorMessage = '';
            this.successMessage = '';
            this.submitting = true;
            
            try {
                const response = await fetch('/api/ratings/submit', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        vod_id: vodId,
                        rating: this.rating,
                        comment: this.comment.trim()
                    })
                });
                
                const data = await response.json();
                
                if (data.code === 200) {
                    this.successMessage = '评分提交成功！';
                    this.userRating = data.data;
                    
                    
                    setTimeout(() => {
                        this.$root.showRating = false;
                        
                        location.reload();
                    }, 2000);
                } else {
                    this.errorMessage = data.message || '评分提交失败';
                }
            } catch (error) {
                console.error('提交评分失败:', error);
                this.errorMessage = '网络错误，请稍后重试';
            } finally {
                this.submitting = false;
            }
        },
        
        
        async deleteRating() {
            if (!confirm('确定要删除评分吗？')) {
                return;
            }
            
            try {
                const response = await fetch(`/api/ratings/delete?vod_id=${vodId}`, {
                    method: 'DELETE'
                });
                
                const data = await response.json();
                
                if (data.code === 200) {
                    this.successMessage = '评分已删除';
                    this.userRating = null;
                    this.rating = 0;
                    this.comment = '';
                    
                    setTimeout(() => {
                        this.$root.showRating = false;
                        location.reload();
                    }, 1500);
                } else {
                    this.errorMessage = data.message || '删除失败';
                }
            } catch (error) {
                console.error('删除评分失败:', error);
                this.errorMessage = '网络错误，请稍后重试';
            }
        }
      
    };
}
